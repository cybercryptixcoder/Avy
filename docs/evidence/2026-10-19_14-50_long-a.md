#!/usr/bin/env python3
"""One-shot backfill: copy old_events into events with a contiguous seq block.

Append-only event table. Old id stays the primary key, verbatim, forever --
it is the read-side handle and nothing renumbers it. The six-month old-id
readability guarantee is structural: this script only INSERTs, the old id is
the key, and there is no UPDATE/DELETE path to lose it.

    events(id TEXT PRIMARY KEY, seq INTEGER NOT NULL UNIQUE, ..., ...)

Rerun-safe by collision: if it dies halfway the transaction rolls back; if a
previous run partly landed (it cannot, but), the primary key collides instead
of duplicating. No resume logic.

Run with the app stopped: it takes the single SQLite write lock and reads
max(seq) once, so live writes must not interleave.
"""
import sqlite3
import sys

SOURCE = "old_events"
TARGET = "events"
ORDER_BY = "id"  # fixed insertion order: old id, not created_at


def main(path):
    con = sqlite3.connect(path, isolation_level=None)  # manual transactions
    try:
        con.execute("PRAGMA foreign_keys=OFF")
        # BEGIN IMMEDIATE takes the write lock up front rather than on first write.
        con.execute("BEGIN IMMEDIATE")

        row = con.execute(f"SELECT COALESCE(MAX(seq), 0) FROM {TARGET}").fetchone()
        base = row[0]

        old = con.execute(f"SELECT * FROM {SOURCE} ORDER BY {ORDER_BY}").fetchall()
        cols = [d[0] for d in con.execute(f"SELECT * FROM {SOURCE} LIMIT 0").description]
        if "id" not in cols:
            raise SystemExit(f"{SOURCE} has no id column; refusing to guess")
        width = len(cols)
        n = len(old)

        # seq = base+1 .. base+n, contiguous, above every existing row.
        rows = [tuple(r) + (base + i + 1,) for i, r in enumerate(old)]
        placeholders = ", ".join(["?"] * (width + 1))
        con.executemany(
            f"INSERT INTO {TARGET} ({', '.join(cols)}, seq) VALUES ({placeholders})",
            rows,
        )

        # Checks inside the transaction: a failed assert rolls everything back.
        got = con.execute(f"SELECT COUNT(*) FROM {TARGET}").fetchone()[0]
        assert got == n, f"row count {got} != {n}"
        assert (
            con.execute(f"SELECT COUNT(*) FROM {TARGET} WHERE seq < 1 OR seq > {base + n}").fetchone()[0] == 0
        ), "seq escaped the block"
        gaps = con.execute(
            f"""
            SELECT COUNT(*) FROM (
              SELECT seq, LAG(seq) OVER (ORDER BY seq) AS prev FROM {TARGET}
            ) WHERE prev IS NOT NULL AND seq != prev + 1
            """
        ).fetchone()[0]
        assert gaps == 0, f"{gaps} gaps in seq"
        # Every old id resolves.
        missing = con.execute(
            f"SELECT COUNT(*) FROM {SOURCE} o WHERE NOT EXISTS (SELECT 1 FROM {TARGET} t WHERE t.id = o.id)"
        ).fetchone()[0]
        assert missing == 0, f"{missing} old ids unresolved"

        con.execute("COMMIT")
        print(f"ok: {n} rows, seq {base + 1}..{base + n}")
    except BaseException:
        con.execute("ROLLBACK")
        raise
    finally:
        con.close()


if __name__ == "__main__":
    if len(sys.argv) != 2:
        raise SystemExit("usage: backfill.py path/to/eventdb.sqlite3")
    main(sys.argv[1])
